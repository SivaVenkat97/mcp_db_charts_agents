SELECT DATE_TRUNC('month', created_at) as month, COUNT(id) as user_count FROM users WHERE created_at >= NOW() - INTERVAL '1 year' GROUP BY month ORDER BY month;
SELECT tier, COUNT(id) as user_count FROM users WHERE created_at >= NOW() - INTERVAL '1 year' GROUP BY tier;
SELECT vendor, COUNT(DISTINCT user_id) as user_count FROM premium_subscriptions WHERE created_at >= NOW() - INTERVAL '1 year' AND vendor IN ('Stripe', 'PayPal') GROUP BY vendor;
SELECT * FROM users WHERE signup_date >= DATE_SUB(CURDATE(), INTERVAL 1 YEAR)
SELECT * FROM users WHERE signup_date >= DATE('2024-09-26')
SELECT * FROM users WHERE created_at >= (CURRENT_DATE - INTERVAL '1 YEAR')
SELECT tier, COUNT(*) AS total_users FROM users WHERE created_at >= (CURRENT_DATE - INTERVAL '1 YEAR') GROUP BY tier
SELECT vendor, COUNT(*) AS payment_method_count FROM payments_history WHERE user_id IN (SELECT id FROM users WHERE created_at >= (CURRENT_DATE - INTERVAL '1 YEAR')) GROUP BY vendor
SELECT EXTRACT(YEAR FROM created_at) AS year, EXTRACT(MONTH FROM created_at) AS month, COUNT(*) as total_users FROM users WHERE created_at >= NOW() - INTERVAL '1 year' GROUP BY year, month ORDER BY year, month;
SELECT u.tier, COUNT(*) as total_users FROM users u WHERE u.created_at >= NOW() - INTERVAL '1 year' GROUP BY u.tier;
SELECT ph.vendor, COUNT(*) as total_users FROM payments_history ph JOIN users u ON ph.user_id = u.id WHERE u.created_at >= NOW() - INTERVAL '1 year' GROUP BY ph.vendor;
